# M3 媒体(入库/缩略图/时间线/日期相册/直放播放/任务中心)实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让照片/视频在写入完成后自动入库,并在 Web 上以时间线(月/日分组 + 缩略图)浏览、查看原图、直放视频,同时交付日期相册与最小任务中心。

**Architecture:** 事件驱动异步管线 —— 上传完成时入队 `media.scan`(提取 EXIF/尺寸/时长 → 幂等写 `media_assets` → 再入队 `media.thumb`),缩略图按 `version_sha + size` 内容寻址落在 `derived/thumbs`(可再生、天然幂等);读取走会话 Cookie + 服务端 ACL,时间线用 PostgreSQL 递归 CTE 展开"我可读节点集"后再与 `media_assets` join,确保跨用户不可见且分页无空洞。

**Tech Stack:** Java 21 · Spring Boot 3.3.5 · PostgreSQL 16 · Flyway · Spring Data JPA · metadata-extractor 2.19.0(唯一新依赖)· JUnit 5 + Testcontainers · Vue 3 + TypeScript + Vite + Element Plus + Pinia。

**Spec:** `docs/superpowers/specs/2026-09-28-m3-media-design.md`(实现依据;与本计划冲突时以 spec 为准,并回报冲突)

```java
package com.pnas.server.media;

import com.pnas.server.media.domain.MediaKind;

import java.util.Locale;
import java.util.Set;

/** 媒体类型判定(spec §5):扩展名与 MIME 双通道;判定为媒体才入库。 */
public final class MediaTypes {

    /** 可用纯 Java ImageIO 解码的图片格式(M3 可生成缩略图) */
    private static final Set<String> DECODABLE_IMAGE_EXT =
        Set.of("jpg", "jpeg", "png", "gif", "bmp", "tif", "tiff");
    /** 所有识别为图片的格式(含 M3 不生成缩略图的 HEIC/WEBP) */
    private static final Set<String> IMAGE_EXT =
        Set.of("jpg", "jpeg", "png", "gif", "bmp", "tif", "tiff", "heic", "heif", "webp", "avif");
    private static final Set<String> VIDEO_EXT =
        Set.of("mp4", "m4v", "mov", "webm", "mkv", "avi");

    private MediaTypes() {}

    /** 非媒体返回 null。 */
    public static MediaKind classify(String filename, String mimeType) {
        String ext = extension(filename);
        String mime = mimeType == null ? "" : mimeType.toLowerCase(Locale.ROOT);
        if (IMAGE_EXT.contains(ext) || mime.startsWith("image/")) {
            return MediaKind.IMAGE;
        }
        if (VIDEO_EXT.contains(ext) || mime.startsWith("video/")) {
            return MediaKind.VIDEO;
        }
        return null;
    }

    /** 能否生成缩略图:M3 仅支持纯 Java 可解码的图片;其余入库但置 SKIPPED(spec D3)。 */
    public static boolean canThumbnail(MediaKind kind, String filename) {
        return kind == MediaKind.IMAGE && DECODABLE_IMAGE_EXT.contains(extension(filename));
    }

    static String extension(String filename) {
        if (filename == null) {
            return "";
        }
        int dot = filename.lastIndexOf('.');
        if (dot < 0 || dot == filename.length() - 1) {
            return "";
        }
        return filename.substring(dot + 1).toLowerCase(Locale.ROOT);
    }
}
```

```java
package com.pnas.server.media.testfixtures;

import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;
import java.io.ByteArrayOutputStream;
import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;

/** 测试素材:构造可控 EXIF 的 JPEG(不依赖外部工具或二进制 fixture)。 */
public final class TestImages {

    private TestImages() {}

    /** 无 EXIF 的纯 JPEG。 */
    public static byte[] plainJpeg(int width, int height) throws Exception {
        BufferedImage img = new BufferedImage(width, height, BufferedImage.TYPE_INT_RGB);
        for (int x = 0; x < width; x++) {
            for (int y = 0; y < height; y++) {
                img.setRGB(x, y, ((x * 7) << 16) | ((y * 13) << 8) | 0x40);
            }
        }
        var out = new ByteArrayOutputStream();
        ImageIO.write(img, "jpg", out);
        return out.toByteArray();
    }

    /** 带 EXIF 的 JPEG:Orientation + DateTimeOriginal(UTC)+ GPS 经纬度。 */
    public static byte[] jpegWithExif(Instant takenAt, double lat, double lon, int orientation,
                                      int width, int height) throws Exception {
        byte[] base = plainJpeg(width, height);
        byte[] app1 = exifApp1(takenAt, lat, lon, orientation);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        out.write(base, 0, 2);          // SOI
        out.write(app1);                // APP1(EXIF)
        out.write(base, 2, base.length - 2);
        return out.toByteArray();
    }

    private static byte[] exifApp1(Instant takenAt, double lat, double lon, int orientation) {
        String dt = DateTimeFormatter.ofPattern("yyyy:MM:dd HH:mm:ss")
            .withZone(ZoneOffset.UTC).format(takenAt);
        byte[] dtBytes = (dt + "\0").getBytes(StandardCharsets.US_ASCII);   // 20 字节

        // 偏移布局(相对 TIFF 头起点):头 8 + IFD0(2 项:2+2*12+4=30) + ExifIFD(1 项:2+12+4=18) + GPS IFD(4 项:2+48+4=54)
        final int ifd0Offset = 8;
        final int exifIfdOffset = ifd0Offset + 30;
        final int gpsIfdOffset = exifIfdOffset + 18;
        final int dataOffset = gpsIfdOffset + 54;      // 之后依次:dtBytes, lat rationals, lon rationals
        final int latOffset = dataOffset + dtBytes.length;
        final int lonOffset = latOffset + 24;

        ByteBuffer b = ByteBuffer.allocate(dataOffset + dtBytes.length + 48).order(ByteOrder.LITTLE_ENDIAN);
        // TIFF 头
        b.put((byte) 'I').put((byte) 'I').putShort((short) 42).putInt(ifd0Offset);
        // IFD0
        b.putShort((short) 2);
        b.putShort((short) 0x0112).putShort((short) 3).putInt(1).putShort((short) orientation).putShort((short) 0);
        b.putShort((short) 0x8769).putShort((short) 4).putInt(1).putInt(exifIfdOffset);
        b.putInt(0);
        // Exif SubIFD
        b.putShort((short) 1);
        b.putShort((short) 0x9003).putShort((short) 2).putInt(dtBytes.length).putInt(dataOffset);
        b.putInt(0);
        // GPS IFD
        b.putShort((short) 4);
        b.putShort((short) 0x0001).putShort((short) 2).putInt(2).put((byte) (lat >= 0 ? 'N' : 'S')).put((byte) 0);
        b.putShort((short) 0x0002).putShort((short) 5).putInt(3).putInt(latOffset);
        b.putShort((short) 0x0003).putShort((short) 2).putInt(2).put((byte) (lon >= 0 ? 'E' : 'W')).put((byte) 0);
        b.putShort((short) 0x0004).putShort((short) 5).putInt(3).putInt(lonOffset);
        b.putInt(0);
        // 数据区
        b.put(dtBytes);
        putRationals(b, Math.abs(lat));
        putRationals(b, Math.abs(lon));

        byte[] tiff = b.array();
        ByteBuffer app1 = ByteBuffer.allocate(tiff.length + 10).order(ByteOrder.BIG_ENDIAN);
        app1.put((byte) 0xFF).put((byte) 0xE1);
        app1.putShort((short) (tiff.length + 8));       // 长度含自身 2 字节
        app1.put("Exif\0\0".getBytes(StandardCharsets.US_ASCII));
        app1.put(tiff);
        return app1.array();
    }

    /** 度/分/秒 三个 RATIONAL。 */
    private static void putRationals(ByteBuffer b, double deg) {
        int d = (int) deg;
        double minutesFull = (deg - d) * 60;
        int m = (int) minutesFull;
        double sec = (minutesFull - m) * 60;
        b.putInt(d).putInt(1);
        b.putInt(m).putInt(1);
        b.putInt((int) Math.round(sec * 10000)).putInt(10000);
    }
}
```

```java
package com.pnas.server.media;

import com.drew.imaging.ImageMetadataReader;
import com.drew.metadata.Metadata;
import com.drew.metadata.exif.ExifIFD0Directory;

import com.drew.metadata.exif.GpsDirectory;
import com.drew.metadata.jpeg.JpegDirectory;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import java.io.InputStream;
import java.time.Instant;
import java.util.Date;

/** 读取 EXIF(拍摄时间/Orientation/GPS/尺寸);任何解析失败都降级为空结果,不抛异常(spec §7)。 */
@Component
public class ExifReader {

    public record ExifData(Instant takenAt, Integer width, Integer height,
                           Double gpsLat, Double gpsLon, Integer orientation) {
        public static ExifData empty() {
            return new ExifData(null, null, null, null, null, null);
        }
    }

    private static final Logger log = LoggerFactory.getLogger(ExifReader.class);

    public ExifData read(InputStream in) {
        try {
            Metadata md = ImageMetadataReader.readMetadata(in);
            Instant takenAt = null;
            ExifSubIFDDirectory sub = md.getFirstDirectoryOfType(ExifSubIFDDirectory.class);
            if (sub != null) {
                Date original = sub.getDateOriginal();
                if (original != null) {
                    takenAt = original.toInstant();
                }
            }
            Integer width = null;
            Integer height = null;
            Integer orientation = null;
            ExifIFD0Directory ifd0 = md.getFirstDirectoryOfType(ExifIFD0Directory.class);
            if (ifd0 != null) {
                orientation = ifd0.getInteger(ExifIFD0Directory.TAG_ORIENTATION);
                width = ifd0.getInteger(ExifIFD0Directory.TAG_IMAGE_WIDTH);
                height = ifd0.getInteger(ExifIFD0Directory.TAG_IMAGE_HEIGHT);
            }
            JpegDirectory jpeg = md.getFirstDirectoryOfType(JpegDirectory.class);
            if (jpeg != null) {
                if (width == null) {
                    width = jpeg.getImageWidth();
                }
                if (height == null) {
                    height = jpeg.getImageHeight();
                }
            }
            Double lat = null;
            Double lon = null;
            GpsDirectory gps = md.getFirstDirectoryOfType(GpsDirectory.class);
            if (gps != null && gps.getGeoLocation() != null) {
                lat = gps.getGeoLocation().getLatitude();
                lon = gps.getGeoLocation().getLongitude();
            }
            return new ExifData(takenAt, width, height, lat, lon, orientation);
        } catch (Exception e) {
            log.debug("EXIF 解析失败,按无元数据处理: {}", e.getMessage());
            return ExifData.empty();
        }
    }
}
```

```java
package com.pnas.server.media;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import java.io.DataInputStream;
import java.io.IOException;
import java.io.InputStream;

/** 纯 Java 解析 MP4/MOV 的 mvhd 时长(spec D4);不做完整容器解析。 */
@Component
public class Mp4DurationReader {

    private static final Logger log = LoggerFactory.getLogger(Mp4DurationReader.class);
    private static final long MAX_SKIP_BYTES = 64L * 1024 * 1024;

    public Long readDurationMs(InputStream in) {
        try {
            DataInputStream data = new DataInputStream(in);
            long skipped = 0;
            while (skipped < MAX_SKIP_BYTES) {
                long size = Integer.toUnsignedLong(data.readInt());
                byte[] type = new byte[4];
                data.readFully(type);
                String box = new String(type, java.nio.charset.StandardCharsets.US_ASCII);
                long header = 8;
                if (size == 1) {
                    size = data.readLong();
                    header = 16;
                } else if (size == 0) {
                    return null; // 延伸到文件末尾:无法定位后续 box
                }
                if ("moov".equals(box)) {
                    return readMvhd(data);
                }
                long payload = size - header;
                if (payload < 0) {
                    return null;
                }
                data.skipNBytes(payload);
                skipped += size;
            }
            return null;
        } catch (Exception e) {
            log.debug("MP4 时长解析失败(按未知处理): {}", e.getMessage());
            return null;
        }
    }

    private Long readMvhd(DataInputStream data) throws IOException {
        while (true) {
            long size = Integer.toUnsignedLong(data.readInt());
            byte[] type = new byte[4];
            data.readFully(type);
            String box = new String(type, java.nio.charset.StandardCharsets.US_ASCII);
            long header = 8;
            if (size == 1) {
                size = data.readLong();
                header = 16;
            }
            long payload = size - header;
            if ("mvhd".equals(box)) {
                int version = data.readUnsignedByte();
                data.skipNBytes(3);                     // flags
                long timescale;
                long duration;
                if (version == 1) {
                    data.skipNBytes(16);                // creation + modification(64 位)
                    timescale = Integer.toUnsignedLong(data.readInt());
                    duration = data.readLong();
                } else {
                    data.skipNBytes(8);                 // creation + modification(32 位)
                    timescale = Integer.toUnsignedLong(data.readInt());
                    duration = Integer.toUnsignedLong(data.readInt());
                }
                if (timescale <= 0) {
                    return null;
                }
                return duration * 1000L / timescale;
            }
            if (payload < 0) {
                return null;
            }
            data.skipNBytes(payload);
        }
    }
}
```

```java
package com.pnas.server.blobstore;

import com.pnas.server.common.PnasProperties;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;

/** 派生缓存目录(可再生资源):derived/thumbs/<sha 前 2 位>/<versionSha>-<size>.jpg。 */
@Component
public class DerivedStore {

    private final Path thumbsRoot;

    public DerivedStore(PnasProperties props) throws IOException {
        this.thumbsRoot = props.dataDir().resolve("derived").resolve("thumbs");
        Files.createDirectories(thumbsRoot);
    }

    public Path thumbPath(String versionSha, int size) {
        return thumbsRoot.resolve(versionSha.substring(0, 2)).resolve(versionSha + "-" + size + ".jpg");
    }

    public boolean exists(String versionSha, int size) {
        return Files.exists(thumbPath(versionSha, size));
    }

    public InputStream open(String versionSha, int size) throws IOException {
        return Files.newInputStream(thumbPath(versionSha, size));
    }

    public void writeAtomic(String versionSha, int size, byte[] jpeg) throws IOException {
        Path target = thumbPath(versionSha, size);
        Files.createDirectories(target.getParent());
        Path tmp = Files.createTempFile(thumbsRoot, "thumb-", ".tmp");
        try {
            Files.write(tmp, jpeg);
            Files.move(tmp, target, StandardCopyOption.ATOMIC_MOVE, StandardCopyOption.REPLACE_EXISTING);
        } finally {
            Files.deleteIfExists(tmp);
        }
    }
}
```

```java
package com.pnas.server.media;

import com.pnas.server.blobstore.DerivedStore;
import com.pnas.server.media.domain.MediaKind;
import org.springframework.stereotype.Component;

import javax.imageio.IIOImage;
import javax.imageio.ImageIO;
import javax.imageio.ImageWriteParam;
import javax.imageio.stream.MemoryCacheImageOutputStream;
import java.awt.Graphics2D;
import java.awt.RenderingHints;

/** 缩略图生成:解码 → 按 Orientation 旋转 → 长边缩放(不放大)→ JPEG 0.82(spec §5)。 */
@Component
public class ThumbnailService {

    private static final float JPEG_QUALITY = 0.82f;

    private final DerivedStore derived;

    public ThumbnailService(DerivedStore derived) {
        this.derived = derived;
    }

    /** 生成并落盘;已存在返回 false;格式不支持抛 UnsupportedOperationException。 */
    public boolean generate(String versionSha, String filename, InputStream original, int size,
                            Integer orientation) throws IOException {
        if (derived.exists(versionSha, size)) {
            return false;
        }
        if (!MediaTypes.canThumbnail(MediaKind.IMAGE, filename)) {
            throw new UnsupportedOperationException("该格式不支持缩略图: " + filename);
        }
        derived.writeAtomic(versionSha, size, render(original, size, orientation));
        return true;
    }

    /** 渲染为 JPEG 字节。 */
    public byte[] render(InputStream original, int size, Integer orientation) throws IOException {
        BufferedImage src = ImageIO.read(original);
        if (src == null) {
            throw new IOException("图像无法解码(可能损坏或非图片)");
        }
        BufferedImage rotated = rotate(src, orientation == null ? 1 : orientation);
        int longSide = Math.max(rotated.getWidth(), rotated.getHeight());
        double scale = longSide <= size ? 1.0 : (double) size / longSide;
        int targetW = Math.max(1, (int) Math.round(rotated.getWidth() * scale));
        int targetH = Math.max(1, (int) Math.round(rotated.getHeight() * scale));

        BufferedImage out = new BufferedImage(targetW, targetH, BufferedImage.TYPE_INT_RGB);
        Graphics2D g = out.createGraphics();
        try {
            g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
            g.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
            g.drawImage(rotated, 0, 0, targetW, targetH, null);
        } finally {
            g.dispose();
        }
        return toJpeg(out);
    }

    private BufferedImage rotate(BufferedImage src, int orientation) {
        return switch (orientation) {
            case 3, 4 -> transform(src, Math.PI, src.getWidth(), src.getHeight());
            case 5, 6 -> transform(src, Math.PI / 2, src.getHeight(), src.getWidth());
            case 7, 8 -> transform(src, -Math.PI / 2, src.getHeight(), src.getWidth());
            default -> src;
        };
    }

    private BufferedImage transform(BufferedImage src, double radians, int w, int h) {
        BufferedImage out = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
        Graphics2D g = out.createGraphics();
        try {
            g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
            g.transform(AffineTransform.getRotateInstance(radians, w / 2.0, h / 2.0));
            g.drawImage(src, (w - src.getWidth()) / 2, (h - src.getHeight()) / 2, null);
        } finally {
            g.dispose();
        }
        return out;
    }

    private byte[] toJpeg(BufferedImage image) throws IOException {
        var writer = ImageIO.getImageWritersByFormatName("jpg").next();
        var param = writer.getDefaultWriteParam();
        param.setCompressionMode(ImageWriteParam.MODE_EXPLICIT);
        param.setCompressionQuality(JPEG_QUALITY);
        var out = new ByteArrayOutputStream();
        try (var ios = new MemoryCacheImageOutputStream(out)) {
            writer.setOutput(ios);
            writer.write(null, new IIOImage(image, null, null), param);
        } finally {
            writer.dispose();
        }
        return out.toByteArray();
    }
}
```

```java
package com.pnas.server.media;

import com.pnas.common.error.ErrorCode;
import com.pnas.server.common.error.BusinessException;
import com.pnas.server.files.FileVersionRepository;
import com.pnas.server.files.FilesService;
import com.pnas.server.files.NodeRepository;

import com.pnas.server.media.domain.*;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.io.IOException;
import java.io.InputStream;
import java.io.UncheckedIOException;
import java.util.Optional;
import java.util.UUID;

/** 幂等入库:判定类型 → 提取元数据 → upsert media_assets(spec §5)。 */
@Service
public class MediaIngestService {

    public record IngestResult(UUID assetId, MediaKind kind, ThumbStatus thumbStatus) {}

    private final MediaAssetRepository assets;
    private final NodeRepository nodes;
    private final FileVersionRepository versions;
    private final FilesService files;
    private final ExifReader exif;
    private final Mp4DurationReader mp4;

    public MediaIngestService(MediaAssetRepository assets, NodeRepository nodes,
                              FileVersionRepository versions, FilesService files,
                              ExifReader exif, Mp4DurationReader mp4) {
        this.assets = assets;
        this.nodes = nodes;
        this.versions = versions;
        this.files = files;
        this.exif = exif;
        this.mp4 = mp4;
    }

    @Transactional
    public Optional<IngestResult> ingest(UUID nodeId, UUID versionId) {
        Node node = nodes.findById(nodeId).orElseThrow(() -> new BusinessException(
            ErrorCode.NOT_FOUND, HttpStatus.NOT_FOUND, "节点不存在: " + nodeId));
        FileVersion version = versions.findById(versionId).orElseThrow(() -> new BusinessException(
            ErrorCode.NOT_FOUND, HttpStatus.NOT_FOUND, "文件版本不存在: " + versionId));

        MediaKind kind = MediaTypes.classify(node.getName(), version.getMimeType());
        if (kind == null) {
            return Optional.empty();
        }
        ThumbStatus status = MediaTypes.canThumbnail(kind, node.getName())
            ? ThumbStatus.PENDING
            : ThumbStatus.SKIPPED;

        Metadata md = kind == MediaKind.IMAGE ? readImage(versionId) : readVideo(versionId);
        var takenAt = md.takenAt() != null ? md.takenAt() : version.getCreatedAt();
        var source = md.takenAt() != null ? TakenAtSource.EXIF : TakenAtSource.MTIME;

        MediaAsset asset = assets.findByFileVersionId(versionId).orElse(null);
        if (asset == null) {
            asset = assets.save(MediaAsset.create(versionId, nodeId, kind, takenAt, source,
                md.width(), md.height(), md.gpsLat(), md.gpsLon(), md.durationMs(), status));
        } else {
            asset.updateMetadata(takenAt, source, md.width(), md.height(), md.gpsLat(), md.gpsLon(),
                md.durationMs());
            asset = assets.save(asset);
        }
        return Optional.of(new IngestResult(asset.getId(), kind, asset.getThumbStatus()));
    }

    /** 记录元数据的中间载体。 */
    private record Metadata(java.time.Instant takenAt, Integer width, Integer height,
                            Double gpsLat, Double gpsLon, Long durationMs) {}

    private Metadata readImage(UUID versionId) {
        try (InputStream in = files.openVersion(versionId)) {
            var data = exif.read(in);
            return new Metadata(data.takenAt(), data.width(), data.height(), data.gpsLat(),
                data.gpsLon(), null);
        } catch (IOException e) {
            throw new UncheckedIOException("读取图片内容失败", e);
        }
    }

    private Metadata readVideo(UUID versionId) {
        try (InputStream in = files.openVersion(versionId)) {
            return new Metadata(null, null, null, null, null, mp4.readDurationMs(in));
        } catch (IOException e) {
            throw new UncheckedIOException("读取视频内容失败", e);
        }
    }
}
```

```java
/** media.reindex:按 id 顺序分批回填/刷新媒体索引,每批 200,必要时自链式入队下一批。 */
@Component
public class MediaReindexHandler implements JobHandler {

    private static final int BATCH = 200;

    private final NodeRepository nodes;
    private final FileVersionRepository versions;
    private final MediaIngestService ingest;
    private final ObjectProvider<JobService> jobs;

    public MediaReindexHandler(NodeRepository nodes, FileVersionRepository versions,
                               MediaIngestService ingest, ObjectProvider<JobService> jobs) {
        this.nodes = nodes;
        this.versions = versions;
        this.ingest = ingest;
        this.jobs = jobs;
    }

    @Override
    public String type() {
        return "media.reindex";
    }

    @Override
    public Map<String, Object> run(Job job) {
        Object cursor = job.getPayload().get("cursor");
        List<Node> batch = cursor == null
            ? nodes.findTop200ByOrderByIdAsc()
            : nodes.findTop200ByIdGreaterThanOrderByIdAsc(UUID.fromString((String) cursor));

        int scanned = 0;
        for (Node node : batch) {
            if (node.getKind() != Node.Kind.FILE) {
                continue;
            }
            var latest = versions.findTopByNodeIdOrderByVersionNoDesc(node.getId());
            if (latest.isPresent()) {
                ingest.ingest(node.getId(), latest.get().getId());
                scanned++;
            }
        }
        Map<String, Object> out = new HashMap<>();
        out.put("scanned", scanned);
        if (batch.size() == BATCH) {
            String next = batch.get(batch.size() - 1).getId().toString();
            jobs.getObject().enqueue("media.reindex", Map.of("cursor", next), 1, 0);
            out.put("next", next);
        } else {
            out.put("next", null);
        }
        return out;
    }
}
```

```java
/** 时间线读取:可见性由"我的 home + 显式授权节点"向下递归展开决定(spec §5)。 */
@Service
public class MediaTimelineService {

    public record TimelineItem(UUID assetId, UUID nodeId, String name, MediaKind kind, Instant takenAt,
                               Integer width, Integer height, Long durationMs, ThumbStatus thumbStatus,
                               long sizeBytes) {}

    public record PeriodCount(String period, long count) {}

    public record YearMonths(int year, List<PeriodCount> months) {}

    private static final int MAX_LIMIT = 200;

    private final MediaAssetRepository assets;
    private final AclRepository acls;
    private final GroupRepository groups;
    private final NodeRepository nodes;
    private final FilesService files;

    public MediaTimelineService(MediaAssetRepository assets, AclRepository acls, GroupRepository groups,
                                NodeRepository nodes, FilesService files) {
        this.assets = assets;
        this.acls = acls;
        this.groups = groups;
        this.nodes = nodes;
        this.files = files;
    }

    @Transactional(readOnly = true)
    public List<TimelineItem> timeline(SessionPrincipal me, String cursor, int limit, MediaKind kind) {
        int cap = Math.min(Math.max(limit, 1), MAX_LIMIT);
        Instant cursorTakenAt = null;
        String cursorId = null;
        if (cursor != null && !cursor.isBlank()) {
            String[] parts = cursor.split("\\|", 2);
            cursorTakenAt = Instant.parse(parts[0]);
            cursorId = parts[1];
        }
        List<Object[]> rows = assets.timelinePage(readableRoots(me), kind == null ? null : kind.name(),
            cursorTakenAt == null ? null : cursorTakenAt.toString(), cursorId, cap);
        List<TimelineItem> items = new ArrayList<>();
        for (Object[] r : rows) {
            items.add(new TimelineItem((UUID) r[0], (UUID) r[1], (String) r[2],
                MediaKind.valueOf((String) r[3]), ((java.sql.Timestamp) r[4]).toInstant(),
                (Integer) r[5], (Integer) r[6], r[7] == null ? null : ((Number) r[7]).longValue(),
                ThumbStatus.valueOf((String) r[8]), ((Number) r[9]).longValue()));
        }
        return items;
    }

    @Transactional(readOnly = true)
    public List<PeriodCount> buckets(SessionPrincipal me, int year, MediaKind kind) {
        List<PeriodCount> out = new ArrayList<>();
        for (Object[] r : assets.bucketCounts(readableRoots(me), year, kind == null ? null : kind.name())) {
            out.add(new PeriodCount((String) r[0], ((Number) r[1]).longValue()));
        }
        return out;
    }

    @Transactional(readOnly = true)
    public List<YearMonths> albumsDates(SessionPrincipal me) {
        Map<Integer, List<PeriodCount>> byYear = new LinkedHashMap<>();
        for (PeriodCount c : buckets(me, 0, null)) {
            int year = Integer.parseInt(c.period().substring(0, 4));
            byYear.computeIfAbsent(year, y -> new ArrayList<>()).add(c);
        }
        List<YearMonths> out = new ArrayList<>();
        byYear.forEach((year, months) -> out.add(new YearMonths(year, months)));
        return out;
    }

    /** 单条读取:无权或已回收一律 404(不泄露存在性)。 */
    @Transactional(readOnly = true)
    public MediaAsset requireReadable(SessionPrincipal me, UUID assetId) {
        MediaAsset asset = assets.findById(assetId).orElseThrow(() -> notFound());
        Node node = nodes.findById(asset.getNodeId()).orElseThrow(() -> notFound());
        if (node.getTrashedAt() != null) {
            throw notFound();
        }
        if (!files.canRead(me, node.getId())) {
            throw notFound();
        }
        return asset;
    }

    /** 我可读节点根集合:我的 home + 显式授予我或我所在组的节点。 */
    Set<UUID> readableRoots(SessionPrincipal me) {
        Set<UUID> roots = new LinkedHashSet<>();
        roots.add(files.homeOf(me).getId());
        acls.findByPrincipalTypeAndPrincipalId(AclEntry.PrincipalType.USER, me.userId()).stream()
            .filter(MediaTimelineService::grantsRead)
            .forEach(e -> roots.add(e.getNodeId()));
        for (UUID groupId : groups.findGroupIdsByUserId(me.userId())) {
            acls.findByPrincipalTypeAndPrincipalId(AclEntry.PrincipalType.GROUP, groupId).stream()
                .filter(MediaTimelineService::grantsRead)
                .forEach(e -> roots.add(e.getNodeId()));
        }
        return roots;
    }

    /** 仅"含 r 且不含显式拒绝 -"的条目可作为可见性根。 */
    private static boolean grantsRead(AclEntry e) {
        return e.getPerms().indexOf('r') >= 0 && e.getPerms().indexOf('-') < 0;
    }

    private BusinessException notFound() {
        return new BusinessException(ErrorCode.NOT_FOUND, HttpStatus.NOT_FOUND, "媒体不存在");
    }
}
```

```java
package com.pnas.server.files;

import com.pnas.common.error.ErrorCode;
import com.pnas.server.common.error.BusinessException;
import org.springframework.http.HttpStatus;

/** HTTP Range 解析(single range):与 M2 的下载行为完全一致,供节点与媒体端点共用。 */
public record RangeSpec(long start, long length, boolean ranged) {

    public static RangeSpec parse(String header, long totalSize) {
        if (header == null || header.isBlank()) {
            return new RangeSpec(0, totalSize, false);
        }
        if (!header.startsWith("bytes=")) {
            throw new BusinessException(ErrorCode.BAD_REQUEST, HttpStatus.BAD_REQUEST,
                "仅支持 bytes 单位的 Range: " + header);
        }
        String spec = header.substring("bytes=".length()).trim();
        if (spec.contains(",")) {
            throw new BusinessException(ErrorCode.BAD_REQUEST, HttpStatus.BAD_REQUEST,
                "仅支持单段 Range");
        }
        int dash = spec.indexOf('-');
        if (dash < 0) {
            throw new BusinessException(ErrorCode.BAD_REQUEST, HttpStatus.BAD_REQUEST,
                "Range 格式不合法: " + header);
        }
        String first = spec.substring(0, dash).trim();
        String last = spec.substring(dash + 1).trim();
        try {
            long start;
            long end;
            if (first.isEmpty()) {
                long suffix = Long.parseLong(last);
                if (suffix <= 0) {
                    throw outOfRange();
                }
                start = Math.max(0, totalSize - suffix);
                end = totalSize - 1;
            } else {
                start = Long.parseLong(first);
                end = last.isEmpty() ? totalSize - 1 : Long.parseLong(last);
            }
            if (start < 0 || start >= totalSize) {
                throw outOfRange();
            }
            end = Math.min(end, totalSize - 1);
            if (end < start) {
                throw outOfRange();
            }
            return new RangeSpec(start, end - start + 1, true);
        } catch (NumberFormatException e) {
            throw new BusinessException(ErrorCode.BAD_REQUEST, HttpStatus.BAD_REQUEST,
                "Range 数字不合法: " + header);
        }
    }

    private static BusinessException outOfRange() {
        return new BusinessException(ErrorCode.BAD_REQUEST,
            HttpStatus.REQUESTED_RANGE_NOT_SATISFIABLE, "Range 越界");
    }
}
```

```java
    }
```

```java
    public AssetDetail detail(@AuthenticationPrincipal SessionPrincipal me, @PathVariable UUID id) {
        MediaAsset asset = timeline.requireReadable(me, id);
        var node = nodes.findById(asset.getNodeId()).orElseThrow();
        var version = versions.findById(asset.getFileVersionId()).orElseThrow();
        return new AssetDetail(asset.getId(), asset.getNodeId(), node.getName(), asset.getKind().name(),
            asset.getTakenAt(), asset.getTakenAtSource().name(), asset.getWidth(), asset.getHeight(),
            asset.getDurationMs(), asset.getThumbStatus().name(), asset.getGpsLat() != null,
            version.getSizeBytes());
    }

    @GetMapping("/assets/{id}/thumb")
    public ResponseEntity<byte[]> thumb(@AuthenticationPrincipal SessionPrincipal me,
                                        @PathVariable UUID id,
                                        @RequestParam(defaultValue = "256") int size,
                                        @RequestHeader(value = HttpHeaders.IF_NONE_MATCH, required = false) String ifNoneMatch) {
        if (size != 256 && size != 512) {
            throw new BusinessException(ErrorCode.BAD_REQUEST, HttpStatus.BAD_REQUEST,
                "仅支持 256 或 512 缩略图");
        }
        MediaAsset asset = timeline.requireReadable(me, id);
        if (asset.getThumbStatus() != ThumbStatus.READY) {
            throw new BusinessException(ErrorCode.THUMB_UNSUPPORTED, HttpStatus.NOT_FOUND,
                "该媒体暂无可用缩略图");
        }
        var version = versions.findById(asset.getFileVersionId()).orElseThrow();
        String sha = version.getManifestSha256();
        String etag = "\"" + sha + "-" + size + "\"";
        if (etag.equals(ifNoneMatch)) {
            return ResponseEntity.status(HttpStatus.NOT_MODIFIED).eTag(etag).build();
        }
        try (InputStream in = derived.open(sha, size)) {
            return ResponseEntity.ok()
                .eTag(etag)
                .header(HttpHeaders.CACHE_CONTROL, "private, max-age=604800")
                .contentType(MediaType.IMAGE_JPEG)
                .body(in.readAllBytes());
        } catch (IOException e) {
            throw new UncheckedIOException("读取缩略图失败", e);
        }
    }

    @GetMapping("/assets/{id}/content")
    public ResponseEntity<StreamingResponseBody> content(
            @AuthenticationPrincipal SessionPrincipal me,
            @PathVariable UUID id,
            @RequestHeader(value = HttpHeaders.RANGE, required = false) String rangeHeader) {
        MediaAsset asset = timeline.requireReadable(me, id);
        long total = files.sizeOf(asset.getNodeId());
        RangeSpec spec = RangeSpec.parse(rangeHeader, total);
        var content = files.content(me, asset.getNodeId(), spec.start(), spec.length());
        StreamingResponseBody body = out -> {
            try (InputStream in = content.stream()) {
                in.transferTo(out);
            }
        };
        var builder = ResponseEntity.status(spec.ranged() ? HttpStatus.PARTIAL_CONTENT : HttpStatus.OK)
            .header(HttpHeaders.ACCEPT_RANGES, "bytes")
            .contentLength(spec.length());
        if (spec.ranged()) {
            builder.header(HttpHeaders.CONTENT_RANGE,
                "bytes " + spec.start() + "-" + (spec.start() + spec.length() - 1) + "/" + total);
        }
        return builder.body(body);
    }

    @PostMapping("/reindex")
    public ResponseEntity<Map<String, Object>> reindex(@AuthenticationPrincipal SessionPrincipal me) {
        if (!me.isAdmin()) {
            throw new BusinessException(ErrorCode.FORBIDDEN, HttpStatus.FORBIDDEN, "仅管理员可重建媒体索引");
        }
        UUID jobId = jobs.enqueue("media.reindex", Map.of(), 1, 0);
        return ResponseEntity.accepted().body(Map.of("jobId", jobId.toString()));
    }
}
```
